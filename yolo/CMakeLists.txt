if(TARGET alanxinn_cvinfer_cuda)
  set(CVINFER_YOLO_HAS_CUDA_BACKEND ON)
else()
  set(CVINFER_YOLO_HAS_CUDA_BACKEND OFF)
endif()

set(CVINFER_YOLO_SOURCES src/yolo.cpp src/yolo_cuda_loader.cpp)
if(NOT CVINFER_YOLO_HAS_CUDA_BACKEND)
  list(APPEND CVINFER_YOLO_SOURCES src/yolo_cuda_stub.cpp)
endif()

add_library(alanxinn_cvinfer_yolo STATIC ${CVINFER_YOLO_SOURCES})
target_include_directories(alanxinn_cvinfer_yolo PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(alanxinn_cvinfer_yolo PUBLIC alanxinn_cvinfer_core)
if(CVINFER_YOLO_HAS_CUDA_BACKEND)
  target_compile_definitions(alanxinn_cvinfer_yolo PUBLIC CVINFER_HAS_CUDA_BACKEND=1)
  target_link_libraries(alanxinn_cvinfer_yolo PUBLIC alanxinn_cvinfer_cuda)
else()
  target_compile_definitions(alanxinn_cvinfer_yolo PRIVATE CVINFER_CUDA_STATIC)
endif()
if(MSVC AND CVINFER_YOLO_HAS_CUDA_BACKEND)
  target_link_libraries(alanxinn_cvinfer_yolo INTERFACE delayimp)
  set_property(TARGET alanxinn_cvinfer_yolo APPEND PROPERTY
    INTERFACE_LINK_OPTIONS "/DELAYLOAD:alanxinn_cvinfer_cuda.dll")
endif()

option(BUILD_YOLO_UNIFIED_DEMOS "Build demos for the unified yolo11 adapters" ON)
if(BUILD_YOLO_UNIFIED_DEMOS)
  add_executable(alanxinn_yolo11_det_core_demo examples/yolo11_det_core_demo.cpp)
  target_link_libraries(alanxinn_yolo11_det_core_demo PRIVATE alanxinn_cvinfer_yolo alanxinn_cvinfer_common)
  cvinfer_enable_delay_load(alanxinn_yolo11_det_core_demo)
  cvinfer_copy_runtime(alanxinn_yolo11_det_core_demo)

  add_executable(alanxinn_yolo11_seg_core_demo examples/yolo11_seg_core_demo.cpp)
  target_link_libraries(alanxinn_yolo11_seg_core_demo PRIVATE alanxinn_cvinfer_yolo alanxinn_cvinfer_common)
  cvinfer_enable_delay_load(alanxinn_yolo11_seg_core_demo)
  cvinfer_copy_runtime(alanxinn_yolo11_seg_core_demo)
endif()
